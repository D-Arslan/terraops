# Serving API image dependencies — SLIM and CPU-only.
# What the API image deliberately does NOT contain (unlike requirements.txt):
#   matplotlib, seaborn, scikit-learn  -> training/plotting only
#   dvc[s3], tqdm                       -> pipeline/data tooling, not serving
#   streamlit / folium                 -> the UI is a separate, torch-free image
# torch/torchvision come from the PyTorch CPU index (+cpu wheels): no CUDA payload,
# which is the single biggest lever against a multi-GB image.

--extra-index-url https://download.pytorch.org/whl/cpu

torch==2.10.0+cpu
torchvision==0.25.0+cpu
pillow==12.1.1
numpy==2.4.3
pyyaml==6.0.3

# Registry client — loads models:/...@champion via the tracking server's artifact

# Sprint 4 — observability.
# psycopg2-BINARY (not psycopg2): ships a prebuilt wheel, so the API image needs
# no libpq-dev / gcc layer. Prediction logging is append-only and low-volume;
# the binary build's caveats (no custom libpq, threading edge cases at extreme
# scale) do not apply here, and the image stays slim.
psycopg2-binary==2.9.11

# Prometheus exposition on /metrics. In-process counters/histograms only — no
# push gateway, no sidecar: the API is scraped, which is the pull model
# Prometheus is designed around.
prometheus-client==0.25.0
